\chapter{Deterministic and Stochastic Correlation}

\section{Expectation}

\subsection{Non-Equal Likelihood}

Let $X$ be a random variable with a finite number of outcomes $x_i$
for $i=1,\ldots,n$ with probability $p_i$.  The {\bf expectation} 
of $X$ is defined as
\be
\ex \defe \sum_{i=1}^n p_i \; x_i.
\ee
Because the sum of all probabilities is identically equal to unity,
\be
\sum_{i=1}^n p_i = 1.0,
\ee
and the expected
value is considered the {\bf weighted average}, where each $p_i$ is the 
{\bf weight}.

\begin{Hremark}
The weighted average is also known as the {\bf first moment}, which is 
analogous to the {\bf center of mass} of a rigid body.
\end{Hremark}

\subsection{Equal Likelihood}
For the special case where each probability is equally likely, 
\be
p_1 = p_2 = \cdots = p_n = \frac{1}{n},
\label{eq:equal_probability}
\ee
the expectation of $\ex$ has the specialized form $\mu$,
\be
\mu \defe \ex \overset{(\ref{eq:equal_probability})}{=} 
\frac{1}{n} \sum_{i=1}^n \; x_i.
\label{eq:simple_average}
\ee
and is referred to as the {\bf simple average}.

\begin{Hremark}
The expectation $\ex$, 
also known as the weighted average of $X$, should
not be confused with $\mu$, the simple average of $X$.  
The latter is a special
case of the former, as just demonstrated in the above sections.
\end{Hremark}

\begin{Hexample}
{
Consider a 6-sided dice, with faces 1 to 6.  Consider the dice
to be a {\em fair} dice, so that the probability of rolling
a 1 is the same as rolling a 2, and is the same as rolling any 
of the other remaining numbers 3 to 6.   Thus, the fair dice
has equal likelihood of any of the 6 outcomes, and 
Eq.~(\ref{eq:simple_average}) is used.
The expected value is calculated as
\be
\mu = \ex = \frac{1}{6} \left(1 + 2 + 3 + 4 + 5 + 6 \right) = 3.5
\ee
}
\end{Hexample}

The 6-sided dice roll in the preceding example can be 
illustrated through numerical simulation, implemented in Python
below.  

\begin{minted}
[
frame=lines,
framesep=2mm,
baselinestretch=1.2,
bgcolor=hivory,
fontsize=\footnotesize,
linenos
]
{python}
import matplotlib.pyplot as plt
from matplotlib import rc
rc('font',**{'family':'serif','serif':['Computer Modern Roman']})
rc('text', usetex=True)
import numpy as np
from random import randint  # random integer

n_rolls = 40  # number of dice rolls
roll_number = np.arange(1, n_rolls + 1)
roll_history = np.zeros(n_rolls)  # each roll result recorded in time
average = np.zeros(n_rolls)  # initialize the average, updated after each roll

for roll in roll_number:
    roll_result = randint(1,6)  # roll a fair dice, values [1, 6] 
    print(f'Dice roll number {roll} roll value is {roll_result}.')
    roll_history[roll-1] = int(roll_result)
    sum_after_roll = sum(i for i in roll_history)
    average[roll-1] = sum_after_roll / roll

fig = plt.figure(figsize=(8, 4))  # 8 inches wide, 4 inches tall

ax = fig.add_subplot(1, 1, 1)
ax.plot(roll_number, roll_history, 's', color='blue', label='roll value')
ax.plot(roll_number, average, color='red', marker='o', alpha=0.5, 
        label='roll average')
ax.grid()
ax.set_xlabel("roll number")
ax.set_ylabel("roll value and average")
ax.set_ylim(0, 7)
ax.legend(loc='upper right')
plt.show()
figure_name = "dice_roll"
fig.savefig(figure_name + ".pdf", bbox_inches='tight')


\end{minted}
