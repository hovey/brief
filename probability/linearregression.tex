\section{Linear Regression}

\subsection{Univariate Linear Regression}

\subsubsection{Model}

Univariate linear regression is a linear relationship 
between an independent variable $x \in \real$, 
also called the 
{\em explanatory variable}, and a dependent variable
$y \in \real$, also called a {\em response variable}.

The {\bf model function} is
\be
y = w x + b
\label{eq:lin_reg_model}
\ee
where
\begin{itemize}
\item $w \in \real$ is the slope (also called {\bf weight}) 
acting on the independent variable, and
\item $b \in \real$ is the intercept (also called {\bf bias}).
\end{itemize}
The two real-valued numbers ($w, b$) are called
{\bf model parameters}.

\subsubsection{Data}

Assume $n$ pair of observations ($x_i, y_i$), $i=1\ldots n$, 
have been made.  
Assume equal likelihood of each observation.
Given the observations, the objective
of univariate linear regression is to 
\begin{enumerate}
\item find {\bf best fit model parameters 
$(\hat{w}, \hat{b})$}, and
\item quantify the {\bf correlation coefficient}, 
which is an objective measure of 
how well the best fit model describes the underlying 
observed data.  
\end{enumerate}

\subsubsection{Error Function}
 
For each observation ($x_i, y_i$) pair, there is an error
$e_i(w,b)$ that is defined as the difference between the 
actual observed response $y_i$ and the model's 
predicted response $w x_i + b$, defined as
\be
 e_i(w,b) \defe y_i - \left(w x_i + b \right).
\label{eq:L1_norm_lin_reg}
\ee
The error accumulated for all $n$ observations can be 
written as
\be
\parallel \ve{e} (w, b)
\parallel_1 \;  \defe \sum_{i=1}^n \abs{e_i(w,b)}.
\ee
This is the L1-norm of the error vector $\ve{e}
= [\;e_1, e_2, \ldots, e_n\;] \in \real^n$.
See Section~\ref{sec:pnorm} for more details on p-norms.

The model parameters $(w, b)$ are said to be best fit
when the error is minimized.  While the L1-norm error
in~(\ref{eq:L1_norm_lin_reg}) may be used as the objective
error function, the L2-norm is frequently used as an 
alternative to the L1-norm.

\be
\parallel \ve{e} (w, b)
\parallel_2 \;  \defe \sum_{i=1}^n \left(e_i(w,b)\right)^2 
= \sum_{i=1}^n \left(
 y_i - (w x_i + b)
\right)^2.
\label{eq:L2_norm_lin_reg}
\ee

\begin{Hremark}
The use of the L2-norm is called {\bf ordinary least squares}
(OLS),
wherein the goal is to minimize the square of the sum of the 
deviations of the observation output $y_i$ and the model's
output $(w x_i + b)$.  This is also called the RSS for
{\bf residual sum of squares}.
\end{Hremark}

\begin{Hremark}
The TSS, {\bf total sum of squares}, gives rise to the 
{\bf R-squared metric}, defined as
\be
R^2 \defe 1 - \frac{\mbox{RSS}}{\mbox{TSS}} = 
1 - 
\frac{
 \dst\sum_{i=1}^n \left(e_i(w,b)\right)^2
}{
 \dst\sum_{i=1}^n (y_i - \bar{y})^2
}
\ee
The square of the Pearson Correlation Coefficient (PCC) is
equal to the R-squared value,
\be
(\rho_{X,Y})^2 = R^2.
\ee
\end{Hremark}


\begin{Hresult}
Given 
a univariate linear regression model and data consisting of
$n$ pair of observations ($x_i, y_i$), $i=1\ldots n$, 
the best fit model parameters 
$(\hat{w}, \hat{b})$ are
\be
\hat{w} = 
 \frac{\dst\sum_{i=1}^n 
  \left(x_i - \bar{x}\right)
  \left(y_i - \bar{y}\right)}
  {\dst\sum_{i=1}^n
  \left(x_i - \bar{x}\right)^2}.
  \label{eq:result_w_hat}
\ee 
and
\be
\hat{b} = 
\bar{y} - \hat{w} \; \bar{x}.
\label{eq:result_b_hat}
\ee
\end{Hresult}

\begin{Hproof}
For this model, start with the L2-error norm, defined as
\be
\parallel \ve{e} (w, b)
\parallel_2 \;  \defe \sum_{i=1}^n \left(e_i(w,b)\right)^2 
= \sum_{i=1}^m \left(
 y_i - w x_i - b
\right)^2.
\ee
Using the notation $E_2 \in \real$ to denote the L2-error norm
in~(\ref{eq:L2_norm_lin_reg}),
we expand and simplify the squared term,
\be
E_2(w,b) = 
\sum_{i=1}^n y_i^2 - 2 y_i w x_i - 2 y_i b + w^2 x_i^2 
+ 2 w x_i b + b^2.
\label{eq:E_2_w_b}
\ee
Since $E_2$ is the summation of squared quantities, 
$E_2(w,b) \geq 0\;\forall\; w \in \real, b \in \real$.
$E_2(w,b)$ 
behaves as a quadratic, scalar function.
Values of $(w, b)$ that minimize the error $E_2$ are 
found by taking partial derivatives of the error function 
with respect to the parameters $w$ and $b$:
\be
\frac{\partial E_2}{\partial w} = 
\sum_{i=1}^n -2 y_i x_i + 2 w x_i^2 + 2 x_i b
\label{eq:partial_E_w}
\ee
and
\be
\frac{\partial E_2}{\partial b} = 
\sum_{i=1}^n -2 y_i + 2 w x_i + 2 b.
\label{eq:partial_E_b}
\ee
Setting~(\ref{eq:partial_E_b}) equal to zero, and 
using the definitions of $\bar{x}$ and $\bar{y}$
from~(\ref{eq:simple_average}),
\be
\hat{b} = \bar{y} - w \bar{x} \; \mbox{ to minimize } E_2.
\label{eq:b_for_E_2_min}
\ee
Substituting
(\ref{eq:b_for_E_2_min}) into 
(\ref{eq:E_2_w_b}) for values of $\hat{b}$ that minimize
$E_2$ allows
(\ref{eq:E_2_w_b}) to be written as a function of 
$w$ alone, 
\be
E_2(w) = \sum_{i=1}^n 
\; y_i^2 - 2 y_i w x_i - 2 y_i (\bar{y} - w \bar{x}) + 
w^2 x_i^2 + 
 2 w x_i (\bar{y} - w \bar{x}) + 
(\bar{y} - w \bar{x})^2.
\ee
Then, 
\be
\frac{d E_2}{dw} = 
\sum_{i=1}^n 
\; -2 y_i x_i + 2 y_i \bar{x} + 2 w x_i^2 + 
2 x_i \bar{y} - 4 w x_i \bar{x} +
2(\bar{y} - w \bar{x})(-\bar{x})  
\overset{\mbox{\tiny set}}{=} 0, 
\ee
or
\be
0  = \sum_{i=1}^n 
-y_i x_i + y_i \bar{x} + w x_i^2 + 
 x_i \bar{y} - 2 w x_i \bar{x} + w \bar{x} \bar{x} - 
 \bar{x} \bar{y}, 
\ee
\be
 0 = \sum_{i=1}^n
 -\left(x_i - \bar{x}\right)
  \left(y_i - \bar{y}\right) 
  + 
  w \left(x_i - \bar{x}\right)^2.
%\end{align}
\ee
Solving for $w$ is then the best fit $\hat{w}$
\be
\hat{w} = \frac{\dst\sum_{i=1}^n 
  \left(x_i - \bar{x}\right)
  \left(y_i - \bar{y}\right)}
  {\dst\sum_{i=1}^n
  \left(x_i - \bar{x}\right)^2}.
\ee
Multiplying both the numerator and the denominator
by $\frac{1}{n}$ gives
\be
\hat{w} = \frac{\cov(X,Y)}{\var(X)}.
\label{eq:w_cov_var}
\ee
Thus the first objective of univariate linear regression, 
find model parameters ($\hat{w}, \hat{b}$) 
the best fit to the observed data (by minimizing
the error between the model and the data) has been accomplished.
\end{Hproof}
The second and final objective, to quantify the quality of
the best fit, follows.
\begin{Hresult}
\be
\rho_{X,Y} = \frac{\hat{w} \; \sigma_x}{\sigma_y}.
\ee
\end{Hresult}

\begin{Hproof}
Using the sample correlation coefficient
$\rho_{X,Y}$, the sample (uncorrected) standard 
deviation $\sigma_x$ and $\sigma_y$, the sample
variance and covariance,
\begin{align}
\var(X) & = \sigma_x^2, \\
\cov(X,Y) &= \rho_{X,Y} \sigma_x \sigma_y,
\end{align}
then
\be
\hat{w} = \frac{\rho_{X,Y} \sigma_y}{\sigma_x} \implies
\rho_{X,Y} = \frac{\hat{w} \; \sigma_x}{\sigma_y}.
\ee
\end{Hproof}

\begin{Hresult}
As long as the univariate linear regression model includes the 
intercept (bias) term $b$ (\ie the 
model is not forced to go through the origin by virtue of 
$b \overset{\mbox{\tiny set}}{=} 0$.), then 
\begin{enumerate}
\item the sum of the errors is zero
\be
\sum_{i=1}^n e_i = 0, \;\; \mbox{and}
\ee
\item the model regression line goes through the center of 
mass point ($\bar{x}, \bar{y}$).
\end{enumerate}
\end{Hresult}

\begin{Hproof}
For part 1, 
start from the definition~(\ref{eq:L1_norm_lin_reg}),
\be
e_i = y_i - w x_i - b,
\ee
add up all the errors, $i=1\ldots n$, and divide by $n$,
\be
  \frac{1}{n} \sum_{i=1}^n e_i  = 
  \frac{1}{n} \sum_{i=1}^n y_i - w x_i - b.
  \ee
Substitute the definitions of $\bar{x}$ and $\bar{y}$
from~(\ref{eq:simple_average})
  \be
  = \bar{y} - w \bar{x} - b \ee
and finally, substitute the model 
equation~(\ref{eq:lin_reg_model}) with values of
$y \mapsto \bar{y}$ and $x \mapsto \bar{x}$,
\be = \bar{y} - w \bar{x} - \bar{y} + w \bar{x} = 0 \ee
which is the result for part 1.
For part 2, show that the error at the point ($\bar{x}, \bar{y}$) 
is zero, 
\be
\left.e_i\right|_{(\bar{x},\bar{y})}
 = \left.y_i\right|_{\bar{y}} - 
(w \left.x_i\right|_{\bar{x}} + b) = \frac{1}{n}\sum_{i=1}^n y_i 
 - \left( 
 \frac{w}{n}\sum_{i=1}^n x_i + b
 \right).
\ee
Rearranging, 
\be
\left.e_i\right|_{(\bar{x},\bar{y})}
 = 
 \frac{1}{n}\sum_{i=1}^n 
 \left(
 y_i - (w x_i + b)
 \right)
 = 
 \frac{1}{n}\sum_{i=1}^n
 e_i = 0
\ee
by the result in part 1.  This 
shows that the point ($\bar{x}, \bar{y}$) is on the 
line described by the model in~(\ref{eq:lin_reg_model}).
\end{Hproof}

\begin{Hexample}
{
%\colorbox{hlightgray}{Here we present a small example of artificially}
Here we present a small example of artificially 
constructed data
useful for purposes of illustration and unit
testing.\footnote{Adapted from http://onlinestatbook.com/2/regression/intro.html}
Consider the following $n=5$ pair of observations
($x_i, y_i$), $i=1\ldots n$,
\begin{align}
\vx & = {\tt [\; 1.00, 2.00, 3.00, 4.00, 5.00 \;]} \\
\vy & = {\tt [\; 1.00, 2.00, 1.30, 3.75, 2.25 \;]}
\end{align}
Calculating the averages, 
\begin{align}
\bar{x} &= {\tt (1.00+2.00+3.00+4.00+5.00)/5 = 3.00 } \\
\bar{y} &= {\tt (1.00+2.00+1.30+3.75+2.25)/5 = 2.06 } 
\end{align}
Next we calculate the population variances and covariance.
Technically, we should use
the {\em sample} coviariance and {\em sample} variance 
in place of the {\em population} formulae.
The difference in the factor of $1/n$ for the population and
$1/(n-1)$ for the sample appear in both the numerator and
denominator for the calculations of $\hat{w}$, $\hat{b}$ and 
$\rho_{X,Y}$, and thus they cancel out, making the calculations
independent of the use of sample or population measures.
\begin{align}
\var(\vx) 
&= {\tt \left((1.00-3.00)^2 + \right.} \nonumber \\
& \hspace{0.7cm} {\tt (2.00-3.00)^2 + } \nonumber \\
& \hspace{0.7cm} {\tt (3.00-3.00)^2 + } \nonumber \\ 
& \hspace{0.7cm} {\tt (4.00-3.00)^2 + } \nonumber \\
& \hspace{0.7cm} {\tt \left.(5.00-3.00)^2)\right)/5
  = 2.00}  
\end{align}
\begin{align}
\var(\vy) 
&= {\tt \left((1.00-2.06)^2 + \right.} \nonumber \\
& \hspace{0.7cm} {\tt (2.00-2.06)^2 + } \nonumber \\
& \hspace{0.7cm} {\tt (1.30-2.06)^2 + } \nonumber \\ 
& \hspace{0.7cm} {\tt (3.75-2.06)^2 + } \nonumber \\
& \hspace{0.7cm} {\tt \left.(2.25-2.06)^2)\right)/5
  = 0.9194} 
\end{align}
\begin{align}
\cov(\vx, \vy) &= {\tt \left((1.00-3.00)(1.00-2.06) + \right.} \nonumber \\
& \hspace{0.7cm} {\tt (2.00-3.00)(2.00-2.06) + } \nonumber \\
& \hspace{0.7cm} {\tt (3.00-3.00)(1.30-2.06) + } \nonumber \\ 
& \hspace{0.7cm} {\tt (4.00-3.00)(3.75-2.06) + } \nonumber \\
& \hspace{0.7cm} {\tt \left.(5.00-3.00)(2.25-2.06)\right)/5
  = 0.850} 
\end{align}
Then, with (\ref{eq:result_w_hat}), or 
equivalently (\ref{eq:w_cov_var}),
\be
\hat{w} = \frac{\cov(\vx, \vy)}{\var(\vx)} = \frac{0.850}{2.00} = \colorbox{yellow}{0.425}
\ee
as the slope (weight), and with (\ref{eq:result_b_hat})
\be
\hat{b} = \bar{y} - \hat{w} \bar{x} = 2.06 - 0.425 * 3.00 = 
\colorbox{yellow}{0.785}
\ee
as the intercept (bias).
The Pearson correlation coefficient (PCC) is found 
with (\ref{eq:pearson_corr_coef}) as,
\begin{align}
\rho_{X,Y} 
&= \frac{\hat{w}\sigma_x}{\sigma_y} = 
\frac{\hat{w} \sqrt{\var(\vx)}}{\sqrt{\var(\vy)}} = 
\frac{\cov(\vx,\vy)}{\sqrt{\var(\vx)}\sqrt{\var(\vy)}} \nonumber \\
&=
\frac{0.425 * \sqrt{2.00}}{\sqrt{0.9194}} = 
\frac{0.850}{\sqrt{2.00}\sqrt{0.9194}} = 
\colorbox{yellow}{0.627}
\end{align}
}
\end{Hexample}

Following is a Python implementation of this example, and the
resulting graphical output, showing the obversed data, 
the mean and variance values, and the linear regression
relationship.

#!/usr/bin/env python3
# lin_regress_1.py
import numpy as np
import matplotlib as mpl
from matplotlib import rc
import matplotlib.pyplot as plt
from matplotlib.ticker import AutoMinorLocator, MultipleLocator, FuncFormatter
rc('font', **{'family': 'serif', 'serif': ['Computer Modern Roman']})
rc('text', usetex=True)

lw = 1 # line width for leader lines

x = np.array([1.00, 2.00, 3.00, 4.00, 5.00])
y = np.array([1.00, 2.00, 1.30, 3.75, 2.25])
theta = np.linspace(0, 2*np.pi, 100)

x_ave = np.mean(x)
y_ave = np.mean(y)

x_var = np.var(x)
y_var = np.var(y)

# the [0,1] component of the covariance matrix
xy_cov = np.cov(x, y, bias=True)[0,1]  

w_hat = xy_cov / x_var
b_hat = y_ave - w_hat * x_ave
x_lin = np.linspace(0, 6, 2)
y_lin = w_hat * x_lin + b_hat  # y = m x + b

dx = dy = 0.15
xmax = 6
ymax = 6
fig = plt.figure(figsize=(xmax, ymax))  # (width, height) inches 
ax = fig.add_subplot(1, 1, 1)
ax.plot(x, y, 's', color='blue', label='observation')
ax.plot(x_ave, y_ave, 'o', color='red', alpha=0.5,
 	label=r'($\bar{x},\bar{y}$) = (' + 
	'{:.2f}'.format(x_ave) + ', ' + 
	'{:.2f}'.format(y_ave) + ')')
ax.plot(x_var * np.cos(theta) + x_ave, 
	y_var * np.sin(theta) + y_ave, 
	color='green', linewidth=1.0, 
	label=r'$\sigma_x^2, \sigma_y^2$ = ' + 
	'{:.2f}'.format(x_var) + ', ' + 
	'{:.4f}'.format(y_var))
ax.plot(x_lin, y_lin, color='red', linewidth=1, 
	label=r'$y = \hat{w} x + \hat{b} = $ ' + 
	'{:.3f}'.format(w_hat) + ' $x\;+\;$' +
	'{:.3f}'.format(b_hat) )
ax.axis('equal')
ax.grid()
ax.set_xlabel('explanatory variable $x$')
ax.set_ylabel('response variable $y$')
ax.set_xlim(0 - dx, xmax + dx)
ax.set_ylim(0 - dy, ymax + dy)
ax.legend(loc='upper left')

plt.show()

figure_name = 'lin_regress_1'
fig.savefig(figure_name + '.pdf', bbox_inches='tight')
\end{minted}
