Following is a list of notation and a brief description, grouped by topic.

\noindent\subsubsection*{Tensors}

\begin{description}[noitemsep,align=right,labelwidth=1.5cm,labelsep=0.8cm]
\item [$a$] zeroth order (scalar)
\item 
\item [$\vA$] first order (vector) in the reference configuration
\item [$A_I$] first order (vector) components in the reference configuration
\item [$\va$] first order (vector) in the current configuration
\item [$a_i$] first order (vector) components in the current configuration
\item 
\item [$\tAbold$] second order (matrix) in the reference configuration
\item [$\tabold$] second order (matrix) in the current configuration
\item [$\tA_{IJ}$] second order (matrix) components in the reference configuration
\item [$\ta_{ij}$] second order (matrix) components in the current configuration
\item 
\item [$\fotA$] fourth order in the reference configuration
\item [$\fotA_{IJKL}$] fourth order components in the reference configuration
\item [$\fota$] fourth order in the current configuration
\item [$\fota_{ijkl}$] fourth order components in the current configuration
\item 
\item [$\vEsubI$] basis vector for the reference (material) configuration
\item [$\vesubi$] basis vector for the current (spatial) configuration
%\item [$\vEsubI$] unit basis vector in the body $B$ reference frame
\item 
\item [$O$] origin of body $B$ and location of the body reference frame $\vEone, \vEtwo, \vEthree$
\item [$o$] origin in spatial configuration
\end{description}

\noindent\subsubsection*{Motion}
\begin{description}[noitemsep,align=right,labelwidth=1.5cm,labelsep=0.8cm]
\item [$\vX$] position of material particle in reference configuration
\item [$\vx$] position of material particle in current configuration
\item [$\vU$] displacement vector from $\vX$ to $\vx$, objective
\item [$\vvarphi$] motion of material particle $\vX$ 
\item 
% \item [$\ve{\varphi}^{-1}_*(\;\cdot\;)$] pull back operator
\item [$\pullback{\cdot}$] pull back operator, $\pullback{\cdot} \defe \tFbold\transpose \;[\;\cdot\;]\; \tFbold$
% \item [$\ve{\varphi}_*(\;\cdot\;)$] push forward operator
\item [$\pushforward{\bullet}$] push forward operator, $\pushforward{\bullet} \defe \tFbold\minustranspose \;[\;\bullet\;]\; \tFbold\inverse$
\item
\item [$\lie{\cdot}$] Lie derivative, 
    $\lie{\cdot} \defe \pushforward{\dst\frac{\partial}{\partial t} \pullback{\cdot}} = \tFbold\minustranspose \; \dst\frac{\partial}{\partial t} 
\left( 
\tFbold\transpose \;[\; \cdot \;]\; \tFbold
\right) \tFbold\inverse$
\item 
\item [$\vv$] velocity of material particle in current 
configuration, non-objective
\item [$\va$] acceleration of material particle in current 
configuration, non-objective
\item 
\item [$\Grad (\;\bullet\;)$] gradient operator in the reference configuration
\item [$\grad (\;\cdot\;)$] gradient operator in the current configuration
\item 
\item [$\Grad \vU$] displacement gradient in the 
reference configuration, 
$\Grad \vU = \tFbold - \vid$
\item [$\grad \vu$] displacement gradient in the 
current configuration, 
$\grad \vu = \vid - \tFbold\inverse$
\item 
\item [$\tFbold \defe \Grad \vvarphi$] deformation gradient,
$\tFbold = \tvbold \tRbold = \tRbold \tUbold$, non-symmetric, 
two-point tensor, objective (transforms as a vector, not a tensor)
\item [$J$] Jacobian of the deformation gradient, 
$J \defe \det \tFbold$
\item 
\item [$\tRbold$] rotation, objective (transforms as a vector, 
not a tensor)
\item [$\tvbold$] left-stretch, objective
\item [$\tUbold$] right-stretch, objective
\end{description}

\subsubsection*{Strain}

\begin{description}[noitemsep,align=right,labelwidth=1.5cm,labelsep=0.8cm]
\item [$\tCbold$] right Cauchy-Green, 
$\tCbold \defe \tFbold\transpose\tFbold$, symmetric, positive definite, objective
\item [$\tbbold$] left Cauchy-Green, 
$\tbbold \defe \tFbold \tFbold \transpose$, symmetric, positive definite, objective
\item 
\item [$\tEbold$] Green-Lagrange, 
$\tEbold \defe \half \left(\tCbold - \vid \right)$,
objective
\item [$\tebold$] Euler-Almansi,
$\tebold \defe \half \left(\vid - \tbbold\inverse \right)$,
objective
\item
\item [$\tEbold_{(m)}$] Seth-Hill
\item [$\lambda_I$] principal stretch ratio
\item
\item [$\ve{\epsilon}$] infinitesimal
\item
\item [$\dot{\tFbold}$] material velocity gradient
\item
\item [$\grad \vv$] spatial velocity gradient, non-symmetric, 
non-objective, covariant
\item 
\item [$\tdbold$] stretch rate (spatial rate of deformation, 
rate of stretch),
\item [] symmetric, objective, covariant,
$\tdbold = \half\left(\grad\vv + \grad\vv\transpose\right)$
\item
\item [$\twbold$] rotation rate (spin, rate of rotation, 
vorticity), 
\item [] skew-symmetric, non-objective, covariant,
$\twbold = \half\left(\grad\vv - \grad\vv\transpose\right)$
\item
\item [$\tDbold$] rotated rate of deformation,
$\tDbold = \tRbold\transpose \tdbold \tRbold$
\item [$\ve{\Omega}$] rate of change of the rotation tensor, skew-symmetric,
$\ve{\Omega} \defe \dot{\tRbold} \tRbold\transpose$
\end{description}

\subsubsection*{Stress}

\begin{description}[noitemsep,align=right,labelwidth=1.5cm,labelsep=0.8cm]
\item [$\ve{\sigma}$] Cauchy, symmetric, objective
\item [$\ve{\tau}$] Kirchhoff, 
$\ve{\tau} \defe J \ve{\sigma}$
\item 
\item [$\hat{\ve{\sigma}}$] Co-rotated Cauchy,
$\hat{\ve{\sigma}} \defe \tRbold\transpose \ve{\sigma} \; \tRbold$
\item [$\ve{\Sigma}$] rotated Kirchhoff, 
$\ve{\Sigma} \defe \tRbold\transpose \ve{\tau} \; \tRbold$
\item 
\item [$\tPbold$] Piola-Kirchhoff One, 
\item [] non-symmetric, objective (transforms as
a vector, not a tensor), 
\item [] 
$\tPbold \defe J \; \ve{\sigma}\;\tFbold\minustranspose = \ve{\tau}\; \tFbold\minustranspose = \tFbold \tSbold$
\item 
%\item [$\ve{\Sigma}$] Piola-Kirchhoff Two, objective
\item [$\tSbold$] Piola-Kirchhoff Two, 
\item [] symmetric, objective,
$\tSbold \defe J \; \tFbold\inverse \;\ve{\sigma}\; \tFbold\minustranspose = \tFbold\inverse \;\ve{\tau}\; \tFbold\minustranspose$
\item 
\item [$\tBbold$] Biot (Jaumann)
\item [$\tAbold$] Almansi
\item 
\item [$\dot{\ve{\sigma}}$] Cauchy stress rate, non-objective
\item 
\item [$\overset{\circ}{\ve{\sigma}}$] Truesdell rate of Cauchy stress, objective
\item 
\item [$\overset{\square}{\ve{\sigma}}$] Green-McInnis-Naghdi rate of Cauchy stress, objective
\item 
\item [$\overset{\triangledown}{\ve{\sigma}}$] Jaumann-Zaremba rate of Cauchy stress, objective
\item
\end{description}
